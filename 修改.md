   // 支持的字段类型（全部 22 种）
   type FormFieldType =
     | 'input'          // 文本输入
     | 'textarea'       // 多行文本
     | 'number'         // 数字输入
     | 'password'       // 密码
     | 'select'         // 下拉选择
     | 'multi-select'   // 多选下拉
     | 'virtual-select' // 虚拟滚动下拉（大数据量）
     | 'checkbox'       // 复选框（单个）
     | 'checkbox-group' // 复选框组
     | 'radio'          // 单选框组
     | 'switch'         // 开关
     | 'slider'         // 滑块
     | 'rate'           // 评分
     | 'date'           // 日期选择
     | 'datetime'       // 日期时间选择
     | 'date-range'     // 日期范围
     | 'time'           // 时间选择
     | 'color'          // 颜色选择
     | 'upload'         // 文件上传
     | 'rich-editor'    // 富文本编辑器
     | 'tags-input'     // 标签输入
     | 'custom'         // 自定义渲染

   interface FormFieldOption {
     label: string
     value: string | number
     disabled?: boolean
   }

   interface FormField {
     field: string                    // 字段名（支持嵌套：'address.city'）
     label?: string                   // 标签，为空则不显示标签
     type: FormFieldType
     placeholder?: string
     defaultValue?: any
     disabled?: boolean | string      // string 为条件表达式，如 "values.type === 'A'"
     hidden?: boolean | string        // 条件显示
     required?: boolean
     rules?: ZodRuleConfig            // Zod 校验规则配置
     options?: FormFieldOption[]      // select/radio/checkbox-group 使用
     remote?: {                       // 远程获取选项
       api: string
       labelField?: string            // 默认 'label'
       valueField?: string            // 默认 'value'
       params?: Record<string, any>
     }
     span?: 6 | 8 | 12 | 24          // 24列栅格，默认 24
     props?: Record<string, any>      // 透传给底层组件的额外 props
     renderKey?: string               // type='custom' 时的渲染函数名
     help?: string                    // 字段下方帮助文字
     tooltip?: string                 // label 右侧问号 Tooltip
   }

   interface ZodRuleConfig {
     min?: number                     // 数字最小值 或 字符串最小长度
     max?: number
     minLength?: number
     maxLength?: number
     pattern?: string                 // 正则表达式字符串
     patternMessage?: string          // 正则不匹配时的错误提示
     email?: boolean
     url?: boolean
     custom?: string                  // 自定义校验函数名，映射到外部传入的 validators
   }

   interface FormGroup {
     title?: string                   // 分组标题
     description?: string             // 分组描述
     collapsible?: boolean            // 可折叠，默认 false
     defaultCollapsed?: boolean       // 默认折叠，默认 false
     fields: FormField[]              // 该分组的字段
     columns?: number                 // 该分组的列数（覆盖全局）
   }

   interface ProFormSchema {
     fields?: FormField[]             // 平铺字段（与 groups 二选一）
     groups?: FormGroup[]             // 分组字段
     layout?: 'vertical' | 'horizontal'  // 默认 'vertical'
     labelWidth?: string | number     // horizontal 布局时标签宽度，默认 100
     columns?: number                 // 全局列数（24栅格下的分列，默认 1 即 span=24）
     submitText?: string              // 提交按钮文字，默认"提交"
     resetText?: string               // 重置按钮文字，默认"重置"
     showSubmit?: boolean             // 默认 true
     showReset?: boolean              // 默认 true
     submitPosition?: 'left' | 'center' | 'right'  // 默认 'right'
   }

2. 创建 src/components/pro/ProForm/FormRenderer.tsx（字段类型 → 组件映射核心）：

   Props：
   - field: FormField
   - control: Control（react-hook-form）
   - watchValues: Record<string, any>（当前表单所有值，用于条件判断）
   - renders?: Record<string, (field: FormField, control: Control) => ReactNode>
   - validators?: Record<string, (value: any) => boolean | string>
   - uploadFn?: (file: File) => Promise<string>

   实现：
   - 根据 field.type 渲染对应组件：
     input → AppInput（password 时 type='password'）
     textarea → Textarea（props.rows 控制行数）
     number → NumberField
     select → AppSelect（remote 时配置 remote + onSearch）
     multi-select → AppSelect（multiple=true）
     virtual-select → VirtualSelect
     checkbox → Checkbox + Label
     checkbox-group → CheckboxGroup（多个 Checkbox 水平/垂直排列）
     radio → RadioGroup
     switch → Switch + Label
     slider → Slider（附带当前值显示）
     rate → Rate
     date → AppDatePicker
     datetime → DateTimePicker
     date-range → AppDateRangePicker
     time → TimePicker
     color → ColorPicker
     upload → AppUpload（props 传 accept/maxCount/listType 等）
     rich-editor → RichEditor（uploadFn 传入）
     tags-input → TagsInput
     custom → renders[field.renderKey]
   
   - 条件表达式执行（hidden/disabled）：
     try { return new Function('values', `return ${expr}`)(watchValues) } catch { return false }
   
   - 从 rules 生成 Zod schema：
     根据 ZodRuleConfig 动态构建 z.string().min().max().email() 等链式调用
   
   - 每个字段包裹在 shadcn FormField + FormItem + FormLabel + FormControl + FormMessage 中
   - required=true：label 后显示红色 *
   - tooltip：label 后 HelpCircle icon + Tooltip
   - help：FormDescription 显示帮助文字
