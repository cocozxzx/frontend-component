   // ── 列类型 ──
   type ColumnType = 'text' | 'number' | 'date' | 'datetime' | 'tag' | 'badge' |
                     'image' | 'link' | 'boolean' | 'action' | 'custom'

   interface TableColumn {
     key: string                          // 数据字段名
     title: string                        // 列标题
     type?: ColumnType                    // 列类型，默认 'text'
     width?: number | string              // 列宽
     minWidth?: number
     fixed?: 'left' | 'right'            // 固定列
     sortable?: boolean                   // 可排序
     filterable?: boolean                 // 可筛选
     hidden?: boolean                     // 默认隐藏（列配置中可显示）
     ellipsis?: boolean                   // 超出省略
     align?: 'left' | 'center' | 'right' // 对齐方式，默认 'left'
     copyable?: boolean                   // 可复制
     // type='tag' 时的颜色映射
     tagMap?: Record<string, 'default' | 'success' | 'warning' | 'destructive' | 'info'>
     // type='tag' 时的文字映射
     labelMap?: Record<string, string>
     // type='boolean' 时的显示文字
     booleanMap?: { true: string; false: string }
     // type='date'/'datetime' 时的格式
     dateFormat?: string
     // type='link' 时的链接配置
     linkConfig?: { href: string | ((row: any) => string); target?: string }
     // type='action' 时的操作按钮
     actions?: ActionButton[]
     // type='custom' 时的渲染函数名（字符串，映射到外部传入的 renders 对象）
     renderKey?: string
   }

   interface ActionButton {
     label: string
     type?: 'default' | 'primary' | 'destructive' | 'warning'
     icon?: string                        // lucide icon name
     permission?: string | string[]       // 权限控制
     hidden?: boolean | string            // true 或字段条件表达式，如 "row.status === 'disabled'"
     disabled?: boolean | string          // 同上
     confirm?: {                          // 点击需二次确认
       title: string
       content?: string
       type?: 'default' | 'warning' | 'danger'
     }
     action: string                       // 事件名，对应外部传入的 onAction 回调
   }

   // ── 搜索字段 ──
   type SearchFieldType = 'input' | 'select' | 'date' | 'date-range' |
                          'number-range' | 'radio' | 'checkbox' | 'custom'

   interface SearchField {
     field: string
     label: string
     type: SearchFieldType
     placeholder?: string
     options?: Array<{ label: string; value: string | number }>
     span?: 6 | 8 | 12 | 24               // 24列栅格，默认 8
     defaultValue?: any
     remote?: boolean
     onSearch?: string                      // 远程搜索函数名，映射到外部传入
   }

   // ── 工具栏按钮 ──
   interface ToolbarButton {
     label: string
     type?: 'default' | 'primary' | 'destructive'
     icon?: string
     permission?: string | string[]
     action: string                         // 对应外部 onToolbarAction 回调
   }

   // ── ProTable Schema ──
   interface ProTableSchema {
     columns: TableColumn[]
     searchFields?: SearchField[]
     toolbar?: ToolbarButton[]
     rowKey?: string                        // 默认 'id'
     showIndex?: boolean                    // 显示序号列
     showSelection?: boolean                // 显示复选框列
     showColumnSetting?: boolean            // 显示列配置按钮，默认 true
     showRefresh?: boolean                  // 显示刷新按钮，默认 true
     pagination?: boolean | {
       defaultPageSize?: number
       pageSizeOptions?: number[]
     }
     api?: string                           // 自动请求的 API 路径
     defaultParams?: Record<string, any>    // 默认查询参数
     bordered?: boolean
     striped?: boolean
     size?: 'sm' | 'md' | 'lg'
   }

   Props：

   - onAction?: (action: string, row: any) => void（操作列按钮回调）
   - onToolbarAction?: (action: string, selectedRows: any[]) => void（工具栏按钮回调）
   - renders?: Record<string, (value: any, row: any, index: number) => ReactNode>（custom 列渲染）
   - searchRenders?: Record<string, (field: SearchField) => ReactNode>（custom 搜索字段渲染）

   实现细节（从上到下）：

   【工具栏区】
   左侧：schema.toolbar 中的按钮（ToolbarButton），按权限过滤显示
   右侧固定：刷新按钮（RotateCw icon）+ 列配置按钮（Settings2 icon）

   【搜索栏区】
   schema.searchFields 存在时渲染 SearchBar
   搜索/重置触发 useTable hook 的 search/reset 方法

   【表格区】
   将 schema.columns 转换为 @tanstack/react-table 的 ColumnDef[]：
   - type='text'：直接显示 value，ellipsis=true 时截断 + Tooltip
   - type='number'：千分位格式化
   - type='date'：用 dateFormat 格式化（默认 'yyyy-MM-dd'）
   - type='datetime'：格式化（默认 'yyyy-MM-dd HH:mm:ss'）
   - type='tag'：渲染 Tag 组件，颜色来自 tagMap，文字来自 labelMap 或原值
   - type='badge'：渲染 AppBadge 组件
   - type='image'：渲染小缩略图（24x24），点击打开 ImagePreview
   - type='link'：渲染超链接（a 标签或 Link）

   - type='custom'：调用 renders[column.renderKey]
   - copyable=true：hover 时右侧显示复制图标，点击复制内容

   schema.showIndex=true：第一列为序号列（(page-1)*pageSize + index + 1）
   schema.showSelection=true：复选框列（联动工具栏 selectedRows 数量提示）
